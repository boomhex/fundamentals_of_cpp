The tools and compiler and linker calls I used to create the static library
libindent.a, were:
I used a simple Makefile to compile, link and create the libraries.

# Compiler and flags
CXX      := g++-15
CXXFLAGS := -Wall -Werror -c
AR       := ar
ARFLAGS  := rcs

# Library name
LIBNAME  := libindent.a

# Source and object files
SRCS := indented_cout.cc indent_more.cc indent_less.cc reset_indentation.cc\
step_size.cc step.cc depth.cc step_size.cc step_locked.cc
OBJS := $(SRCS:.cc=.o)

# Default target
all: $(LIBNAME)

# Rule to build the static library
$(LIBNAME): $(OBJS)
	$(AR) $(ARFLAGS) $@ $^

# Rule for building object files
%.o: %.cc
	$(CXX) $(CXXFLAGS) $< -o $@

# Clean build artifacts
clean:
	rm -f $(OBJS) $(LIBNAME)

# Example: build and link a test program with the library
testfile: main.cc $(LIBNAME)
	$(CXX) -Wall -Werror main.cc -L. -lindent -o test_indent
